import { useMemo, useState } from "react";
import PromptChips from "./PromptChips";
import { PROMPTS } from "../data/promptCatalog";

export default function AssistantChat() {
  const prompts = useMemo(() => PROMPTS, []);
  const [messages, setMessages] = useState([
    {
      role: "assistant",
      content:
        "Hi, I’m Olanna’s Health Assistant. Choose a prompt below and I’ll help with general cycle and reproductive health info.",
    },
  ]);
  const [loading, setLoading] = useState(false);

  // Minimal “context” for MVP — later you’ll wire these to your trackers
  const userContext = {
    cycleDay: 14,
    averageCycleLength: 28,
    phase: "Ovulatory", // your Lotus Cycle mapping
    symptoms: ["cramps", "bloating"],
    bleedingLevel: "none",
    painScore: 4,
    onHormonalContraception: false,
    tryingToConceive: false,
  };

  async function runPrompt(promptId) {
    setLoading(true);

    const prompt = prompts.find((p) => p.id === promptId);
    setMessages((m) => [...m, { role: "user", content: prompt.title }]);

    try {
      const res = await fetch("/api/assistant", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ promptId, userContext }),
      });

      if (!res.ok) throw new Error("Request failed");
      const data = await res.json();

      setMessages((m) => [...m, { role: "assistant", content: data.reply }]);
    } catch (e) {
      setMessages((m) => [
        ...m,
        {
          role: "assistant",
          content:
            "Sorry — I couldn’t respond right now. Please try again in a moment.",
        },
      ]);
    } finally {
      setLoading(false);
    }
  }

  return (
    <div
      style={{
        minHeight: "100vh",
        padding: 20,
        color: "white",
        background:
          "linear-gradient(180deg, rgba(255,120,170,1), rgba(255,120,200,1), rgba(245,200,235,1))",
      }}
    >
      <div style={{ maxWidth: 520, margin: "0 auto" }}>
        <h2 style={{ marginTop: 0 }}>Olanna Assistant</h2>

        <div style={{ marginBottom: 14 }}>
          <PromptChips prompts={prompts} onPick={runPrompt} />
        </div>

        <div
          style={{
            borderRadius: 16,
            padding: 14,
            background: "rgba(255,255,255,0.12)",
            backdropFilter: "blur(10px)",
          }}
        >
          {messages.map((msg, idx) => (
            <div key={idx} style={{ marginBottom: 10 }}>
              <div style={{ opacity: 0.75, fontSize: 12 }}>
                {msg.role === "assistant" ? "Olanna" : "You"}
              </div>
              <div style={{ whiteSpace: "pre-wrap", lineHeight: 1.35 }}>
                {msg.content}
              </div>
            </div>
          ))}

          {loading && <div style={{ opacity: 0.8 }}>Thinking…</div>}
        </div>
      </div>
    </div>
  );
}
